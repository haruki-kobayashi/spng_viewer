cmake_minimum_required(VERSION 3.25)
project(SPNGViewerCpp VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_AUTOMOC ON)
option(SPNG_BUILD_TESTS "Build native comparison and developer test programs" OFF)
set(SPNG_TEST_ROOT "${CMAKE_CURRENT_SOURCE_DIR}" CACHE PATH "Synthetic test fixtures and evidence directory")
find_package(Qt6 6.11.2 EXACT REQUIRED COMPONENTS Core Gui Widgets Concurrent)
find_package(OpenCV 5.0.0 EXACT REQUIRED COMPONENTS core imgproc)
add_library(spng_core STATIC src/core.cpp src/ops.cpp src/scanparams.cpp)
target_include_directories(spng_core PUBLIC src ${OpenCV_INCLUDE_DIRS})
target_link_libraries(spng_core PUBLIC Qt6::Core Qt6::Gui ${OpenCV_LIBS})
target_compile_definitions(spng_core PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN)
add_library(spng_ui STATIC src/imageview.cpp src/imageview.h src/ui_generated.cpp)
target_link_libraries(spng_ui PUBLIC spng_core Qt6::Widgets Qt6::Concurrent)
add_library(spng_viewer_lib STATIC src/viewer.h src/viewer.cpp src/viewer_controls.cpp src/viewer_features.cpp src/viewer_params.cpp src/viewer_export.cpp)
target_link_libraries(spng_viewer_lib PUBLIC spng_ui)
qt_add_executable(SPNGViewer WIN32 src/main.cpp src/smoke.cpp)
target_link_libraries(SPNGViewer PRIVATE spng_viewer_lib)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/src/license_resources.qrc")
    set_target_properties(SPNGViewer PROPERTIES AUTORCC ON)
    target_sources(SPNGViewer PRIVATE src/license_resources.qrc)
endif()
foreach(target spng_core spng_ui spng_viewer_lib SPNGViewer)
    if(MSVC)
        target_compile_options(${target} PRIVATE /utf-8 /W4 /fp:precise)
    endif()
endforeach()
if(SPNG_BUILD_TESTS)
    enable_testing()
    find_package(Qt6 REQUIRED COMPONENTS Test)
    add_executable(spng_core_tests tests/core_tests.cpp)
    target_link_libraries(spng_core_tests PRIVATE spng_core)
    add_executable(spng_ui_preview tests/ui_preview.cpp)
    target_link_libraries(spng_ui_preview PRIVATE spng_ui Qt6::Test Qt6::QOffscreenIntegrationPlugin)
    add_executable(spng_viewer_tests tests/viewer_tests.cpp)
    target_link_libraries(spng_viewer_tests PRIVATE spng_viewer_lib Qt6::Test)
    add_executable(spng_storage_probe tests/storage_probe.cpp)
    target_link_libraries(spng_storage_probe PRIVATE spng_core)
    add_executable(spng_processing_benchmark tests/processing_benchmark.cpp)
    target_link_libraries(spng_processing_benchmark PRIVATE spng_core)
    foreach(target spng_core_tests spng_ui_preview spng_viewer_tests spng_storage_probe spng_processing_benchmark)
        target_compile_definitions(${target} PRIVATE SPNG_TEST_ROOT="${SPNG_TEST_ROOT}")
        if(MSVC)
            target_compile_options(${target} PRIVATE /utf-8 /W4 /fp:precise)
        endif()
    endforeach()
    add_test(NAME core COMMAND spng_core_tests)
    add_test(NAME viewer COMMAND spng_viewer_tests)
endif()
